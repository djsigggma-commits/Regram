def short_err(err):
    err = str(err or "ошибка").strip()
    if "Traceback" in err:
        err = [x for x in err.splitlines() if x.strip()][-1]
    if "exec disabled" in err:
        return "exec выключен: включи «Разрешить exec» в настройках моста"
    if err == "bad token":
        return "неверный токен: заново нажми «Подключить Termux» в мосте"
    return err


def vis(s):
    return len(ANSI.sub("", s))


def box(lines, color=ACC):
    w = max(vis(x) for x in lines)
    print(color + "╭" + "─" * (w + 2) + "╮" + R)
    for x in lines:
        print(color + "│" + R + " " + x + " " * (w - vis(x)) + " " + color + "│" + R)
    print(color + "╰" + "─" * (w + 2) + "╯" + R)


def port_open():
    try:
        socket.create_connection((HOST, PORT), timeout=0.5).close()
        return True
    except OSError:
        return False


def quiet(cmd):
    p = subprocess.run(cmd, shell=True, capture_output=True, text=True)
    return p.returncode == 0 and "Error" not in p.stdout + p.stderr

class Bridge:
    def __init__(self):
        self.sock = None
        self.buf = b""
        self.launched = False
        self.lock = threading.RLock()

    def connect(self):
        self.close()
        s = socket.create_connection((HOST, PORT), timeout=3)
        s.settimeout(120)
        self.sock = s
        self.buf = b""

    def close(self):
        if self.sock is not None:
            try:
                self.sock.close()
            except OSError:
                pass
        self.sock = None

    def roundtrip(self, payload):
        self.sock.sendall(payload)
        while b"\n" not in self.buf:
            chunk = self.sock.recv(65536)
            if not chunk:
                raise ConnectionError("closed")
            self.buf += chunk
        line, self.buf = self.buf.split(b"\n", 1)
        return json.loads(line.decode("utf-8"))

    def send(self, cmd, **kw):
        kw.update(cmd=cmd, token=TOKEN, sid=SID)
        payload = (json.dumps(kw, ensure_ascii=False) + "\n").encode("utf-8")
        with self.lock:
            for attempt in (0, 1):
                try:
                    if self.sock is None:
                        self.connect()
                    return self.roundtrip(payload)
                except ConnectionRefusedError:
                    if self.launched or os.environ.get("TGB_AUTOSTART") == "0":
                        raise
                    self.launched = True
                    if not launch_extera():
                        raise
                except OSError:
                    self.close()
                    raise

def call(cb, *args):
    try:
        n = len(inspect.signature(cb).parameters)
    except (TypeError, ValueError):
        n = len(args)
    return cb(*args[:n])


def drop_callbacks(pid):
    for k in [k for k in CALLBACKS if k[0] == pid]:
        CALLBACKS.pop(k, None)


class Events:
    def __init__(self):
        self.sock = None
        self.thread = None

    def alive(self):
        return self.thread is not None and self.thread.is_alive()

    def start(self):
        if self.alive():
            return
        s = socket.create_connection((HOST, PORT), timeout=3)
        s.settimeout(None)
        hello = {"cmd": "subscribe", "token": TOKEN, "sid": SID}
        s.sendall((json.dumps(hello) + "\n").encode("utf-8"))
        self.sock = s
        self.thread = threading.Thread(target=self.loop, args=(s,), daemon=True)
        self.thread.start()

    def loop(self, s):
        buf = b""
        try:
            while True:
                chunk = s.recv(65536)
                if not chunk:
                    return
                buf += chunk
                while b"\n" in buf:
                    line, buf = buf.split(b"\n", 1)
                    try:
                        ev = json.loads(line.decode("utf-8"))
                    except ValueError:
                        continue
                    if "event" in ev:
                        self.dispatch(ev)
        except OSError:
            return

    def dispatch(self, ev):
        kind, pid, iid = ev.get("event"), ev.get("panel"), ev.get("id")
        if kind == "jcall":
            cb = CALLBACKS.get(("j", ev.get("h"), ev.get("method")))
            args = tuple(dec(a) for a in ev.get("args") or [])
        elif kind == "close":
            OPEN.discard(pid)
            cb = CALLBACKS.get((pid, None, "close"))
            drop_callbacks(pid)
            args = ()
        else:
            cb = CALLBACKS.get((pid, iid, kind))
            args = (ev["value"],) if "value" in ev else ()
        if cb is None:
            return
        try:
            call(cb, *args)
        except Exception as e:
            print("\r" + ERR + "✖ callback: " + str(e) + R)

class Cast:
    def __init__(self, kind, value):
        self.kind, self.value = kind, value


class Arr:
    def __init__(self, kind, values):
        self.kind, self.values = kind, list(values)


def enc(v):
    if isinstance(v, JObj):
        return {"$h": v.h}
    if isinstance(v, Cast):
        return {"$cast": v.kind, "v": v.value}
    if isinstance(v, Arr):
        return {"$arr": v.kind, "v": [enc(x) for x in v.values]}
    if isinstance(v, (list, tuple)):
        return [enc(x) for x in v]
    return v


def dec(v):
    if isinstance(v, dict) and "$h" in v:
        return JObj(v["$h"], v.get("t", ""))
    if isinstance(v, list):
        return [dec(x) for x in v]
    return v


class JMethod:
    def __init__(self, obj, name):
        self.obj, self.name = obj, name

    def __call__(self, *args):
        return JAVA.rpc("j_call", h=self.obj.h, name=self.name, args=[enc(a) for a in args])


class JObj:
    __slots__ = ("h", "t", "__weakref__")

    def __init__(self, h, t=""):
        object.__setattr__(self, "h", h)
        object.__setattr__(self, "t", t)

    def __getattr__(self, name):
        if name.startswith("__"):
            raise AttributeError(name)
        return JMethod(self, name)

    def __setattr__(self, name, value):
        JAVA.rpc("j_set", h=self.h, name=name, value=enc(value))

    def __call__(self, *args):
        return JAVA.rpc("j_new", h=self.h, args=[enc(a) for a in args])

    def __str__(self):
        return JAVA.rpc("j_str", h=self.h)

    def __iter__(self):
        return iter(JAVA.rpc("j_list", h=self.h))

    def __repr__(self):
        return "<java %s #%s>" % (self.t or "object", self.h)

    def __del__(self):
        try:
            JAVA.pending.append(self.h)
        except Exception:
            pass


GRAV = {
    "top_start": 51, "top": 49, "top_end": 53, "center": 17,
    "bottom_start": 83, "bottom": 81, "bottom_end": 85,

class Java:
    def __init__(self):
        self.pending = []
        self.meta = None
        self.classes = {}

    def rpc(self, cmd, **kw):
        if self.pending:
            kw["release"], self.pending = self.pending, []
        r = get_bridge().send(cmd, **kw)
        if not r.get("ok"):
            raise JavaError(short_err(r.get("error")))
        return dec(r.get("r"))

    def cls(self, name):
        c = self.classes.get(name)
        if c is None:
            c = self.classes[name] = self.rpc("j_cls", name=name)
        return c

    def new(self, name, *args):
        return self.cls(name)(*args)

    def activity(self):
        return self.rpc("j_activity")

    def info(self):
        if self.meta is None:
            self.meta = check(get_bridge().send("j_info"))
        return self.meta

    def dp(self, value):
        return int(round(value * float(self.info()["density"])))

    def get(self, obj, name):
        return self.rpc("j_get", h=obj.h, name=name)

    def set(self, obj, name, value):
        self.rpc("j_set", h=obj.h, name=name, value=enc(value))

    def const(self, cls_name, field):
        return self.get(self.cls(cls_name), field)

    def str(self, obj):
        return self.rpc("j_str", h=obj.h)

    def list(self, obj):
        return self.rpc("j_list", h=obj.h)

    def array(self, kind, values):
        return Arr(kind, values)

    def i(self, v):
        return Cast("int", v)

    def f(self, v):
        return Cast("float", v)

    def d(self, v):
        return Cast("double", v)

    def l(self, v):
        return Cast("long", v)

    def b(self, v):
        return Cast("boolean", v)

    def listener(self, iface, returns=None, **handlers):
        EVENTS.start()
        proxy = self.rpc("j_listener", iface=iface, methods=sorted(handlers), returns=returns or {})
        for name, fn in handlers.items():
            CALLBACKS[("j", proxy.h, name)] = fn
        return proxy

    def add_view(self, view, x=None, y=None, gravity="top_start", width=None, height=None, keep=False):
        def size(v):
            if v is None:
                return -2
            return -1 if v == "match" else self.dp(v)

        g = GRAV.get(gravity, 51) if isinstance(gravity, str) else int(gravity)
        px = None if x is None else self.dp(x)
        py = None if x is None and y is None else self.dp(y or 0)
        self.rpc("j_add_view", h=view.h, x=px, y=py, gravity=g, w=size(width), ht=size(height), keep=bool(keep))

    def remove_view(self, view):
        self.rpc("j_remove_view", h=view.h)

    def toast(self, text, long=False):
        Toast = self.cls("android.widget.Toast")
        Toast.makeText(self.activity(), str(text), 1 if long else 0).show()

    def vibrate(self, ms=50):
        vib = self.activity().getSystemService("vibrator")
        vib.vibrate(self.cls("java.lang.Long").parseLong(str(int(ms))))

class Panel:
    def __init__(self, title="", pos="bottom_end", width=260, x=None, y=None, keep=False, id=None):
        self.id = id or "p" + os.urandom(3).hex()
        self.opts = {"title": title, "pos": pos, "width": width, "keep": keep}
        if x is not None:
            self.opts.update(x=x, y=y or 0)
        self.items = []
        self.stack = [self.items]
        self.n = 0

    def add(self, kind, event, cb, id, **props):
        self.n += 1
        iid = id or "i%d" % self.n
        item = {"type": kind, "id": iid}
        item.update({k: v for k, v in props.items() if v is not None})
        self.stack[-1].append(item)
        if cb is not None and event:
            CALLBACKS[(self.id, iid, event)] = cb
        return iid

    def text(self, text, id=None, size=None, color=None, mono=None):
        return self.add("text", None, None, id, text=text, size=size, color=color, mono=mono)

    def button(self, text, callback=None, id=None, color=None):
        return self.add("button", "click", callback, id, text=text, color=color)

    def toggle(self, text, callback=None, value=False, id=None):
        return self.add("toggle", "toggle", callback, id, text=text, value=value)

    def slider(self, text, lo, hi, value=None, callback=None, step=1, id=None):
        start = lo if value is None else value
        return self.add("slider", "slide", callback, id, text=text, min=lo, max=hi, value=start, step=step)

    def input(self, hint, callback=None, button="OK", id=None, clear=True):
        return self.add("input", "submit", callback, id, hint=hint, button=button, clear=clear)

    @contextlib.contextmanager
    def row(self):
        group = {"type": "row", "items": []}
        self.stack[-1].append(group)
        self.stack.append(group["items"])
        try:
            yield self
        finally:
            self.stack.pop()

    def on_close(self, callback):
        CALLBACKS[(self.id, None, "close")] = callback
        return self

    def show(self):
        EVENTS.start()
        spec = dict(self.opts)
        spec.update(id=self.id, items=self.items)
        check(get_bridge().send("panel", **spec))
        OPEN.add(self.id)
        return self

    def set(self, id, **props):
        check(get_bridge().send("set", panel=self.id, id=id, **props))

    def close(self):
        get_bridge().send("close", panel=self.id)
        OPEN.discard(self.id)
        drop_callbacks(self.id)

def reply(r):
    if r.get("output"):
        print(TXT + r["output"].rstrip("\n") + R)

def cmd_show(b, rest):
    args = shlex.split(rest)
    opts, words, it = {}, [], iter(args)

    reply(b.send("show", text=" ".join(words), **opts))

def cmd_sh(b, command):
    if not command:
        print(MUT + "использование: !<команда>" + R)
        return
    b.send("clear")
    p = subprocess.Popen(
        command, shell=True, stdout=subprocess.PIPE, stderr=subprocess.STDOUT,
        text=True, errors="replace", bufsize=1,
    )
    try:
        for line in p.stdout:
            line = line.rstrip("\n")
            print(MUT + line + R)
            b.send("append", text=line, pos=state["pos"])
        code = p.wait()
    except KeyboardInterrupt:
        p.terminate()
        code = -2
    b.send("append", text="— exit %d —" % code, pos=state["pos"])
    print(OK + "✔" + R if code == 0 else ERR + "✖ код выхода %d" % code + R)


def cmd_run(b, rest):

def cmd_py(b, code):
    buf = io.StringIO()
    failed = False
    try:
        try:
            node, mode = compile(code, "<tgb>", "eval"), "eval"
        except SyntaxError:
            node, mode = compile(code, "<tgb>", "exec"), "exec"
        with contextlib.redirect_stdout(buf), contextlib.redirect_stderr(buf):
            val = eval(node, NS)
            if mode == "eval" and val is not None:
                print(repr(val))
    except Exception as e:
        failed = True
        buf.write("%s: %s" % (type(e).__name__, e))
    text = buf.getvalue().rstrip("\n")
    if text:
        print((ERR if failed else TXT) + text + R)
    else:
        print(OK + "✔" + R)
    if failed:
        b.send("show", text="✖ " + text, pos=state["pos"], timeout=6)
    elif text and not API_USE.search(code):
        b.send("show", text=text, pos=state["pos"])

class Tee:
    def __init__(self):
        self.buf = io.StringIO()

    def write(self, s):
        self.buf.write(s)
        sys.__stdout__.write(s)
        sys.__stdout__.flush()
        return len(s)

    def flush(self):
        pass


def pick_code(text):
    m = re.search(r"```(?:python|py)?[ \t]*\n(.*?)```", text, re.S)
    return m.group(1).rstrip() if m else None


def plain(text):
    return re.sub(r"```.*?```", "", text, flags=re.S).strip()


def exec_code(b, code):
    tee = Tee()
    err = ""
    try:
        with contextlib.redirect_stdout(tee), contextlib.redirect_stderr(tee):
            exec(compile(code, "<ai>", "exec"), NS)
    except KeyboardInterrupt:
        err = "прервано"
    except Exception as e:
        err = "%s: %s" % (type(e).__name__, e)
    out = tee.buf.getvalue().strip()
    if err:
        print(ERR + "✖ " + err + R)
        b.send("show", text="✖ " + err, pos=state["pos"], timeout=6)
        return
    if out and not API_USE.search(code):
        b.send("show", text=out, pos=state["pos"])
    print(OK + "✔" + R)


def cmd_ai(b, rest):
    rest = rest.strip()
    if rest in ("reset", "сброс"):
        HISTORY.clear()
        print(OK + "✔ " + R + MUT + "диалог очищен" + R)
        return
    if not rest:
        print(MUT + "использование: ai <что сделать>   (ai reset — забыть диалог)" + R)
        return
    print(MUT + "◌ думаю…" + R)
    r = b.send("ai", prompt=rest, history=HISTORY[-10:])
    if not r.get("ok"):
        reply(r)
        return
    text = str(r.get("text", ""))
    HISTORY.extend([{"role": "user", "content": rest}, {"role": "assistant", "content": text}])
    code = pick_code(text)
    note = plain(text)
    if note:
        print(TXT + note + R)
    if code is None:
        if note:
            b.send("show", title="AI", text=note[:600], pos=state["pos"])
        return
    lines = code.split("\n")
    # Show the complete script before approval, including code after line 40.
    for line in lines:
        print(ACC + "▎" + R + MUT + line + R)
    try:
        ans = input("\001" + ACC + "\002Выполнить? [y/N] \001" + R + "\002")
    except (EOFError, KeyboardInterrupt):
        print()
        return
    if ans.strip().lower() in ("y", "yes", "д", "да"):
        exec_code(b, code)

def is_shell(line):
    first = line.split(None, 1)[0]
    if keyword.iskeyword(first):
        return False
    if shutil.which(first):
        return True
    for mode in ("eval", "exec"):
        try:
            compile(line, "<tgb>", mode)
            return False
        except SyntaxError:
            continue
    return True


def dispatch(b, line):
    line = line.strip()
    if not line:
        return
    if line.startswith("!"):
        cmd_sh(b, line[1:].strip())
        return
    word = line.split(None, 1)[0]
    rest = line[len(word):].strip()
    handler = COMMANDS.get(word.lower())
    if handler is not None:
        handler(b, rest)
    elif is_shell(line):
        cmd_sh(b, line)
    else:
        cmd_py(b, line)


def setup_ns(b):
    NS.update(b=b, show=show, notify=notify, clip=clip, tg=tg, Panel=Panel, wait=wait, java=JAVA)


def shell(b):
    try:
        import readline
    except ImportError:
        pass
    prompt = "\001" + ACC + B + "\002tgb \001" + MUT + "\002❯ \001" + R + "\002"
    banner(b)
    while True:
        try:
            line = input(prompt).strip()
        except (EOFError, KeyboardInterrupt):
            print()
            break
        if line in ("exit", "quit", "q"):
            break
        try:
            dispatch(b, line)
        except ConnectionRefusedError:
            offline()
        except KeyboardInterrupt:
            print()
        except Exception as e:
            print(ERR + "✖ " + str(e) + R)


def main():
    b = get_bridge()
    setup_ns(b)
    if len(sys.argv) > 1:
        argv = sys.argv[1:]
        if argv[0] in ("sh", "py", "tg", "ai", "btn", "notify", "copy"):
            line = " ".join(argv)
        else:
            line = " ".join(shlex.quote(a) for a in argv)
        try:
            dispatch(b, line)
            if argv[0] == "btn" and OPEN:
                print(MUT + "кнопки активны, Ctrl+C — выйти" + R)
                wait()
        except ConnectionRefusedError:
            offline()
            sys.exit(1)
        return
    shell(b)


if __name__ == "__main__":
    main()
